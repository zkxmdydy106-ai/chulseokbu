  function pHtml(b, path, O) {
    let v = O[path];
    if (v === undefined && SCH.form === 'common' && TO_SCHOOL.test(b.text)) v = { html: esc(`${SCH.name || '학교'}장 귀하`) };

    if (v === undefined) inner = textHtml(b.text);
    else if (typeof v === 'string') inner = V(v);
    else { if (v.align) align = v.align; if (v.size) size = v.size; inner = v.html !== undefined ? v.html : V(v.text); }
    const st = [`text-align:${align}`, `font-size:calc(var(--fs, 1) * ${(size * 1.333).toFixed(1)}px)`];
    if (b.bold) st.push('font-weight:700');
    if (b.left) st.push(`padding-left:${(+b.left).toFixed(1)}px`);
    if (b.color && b.color !== '#000000' && v === undefined) st.push(`color:${b.color}`);
    return `<div class="hp${b.serif ? ' serif' : ''}" style="${st.join(';')}">${inner || '&nbsp;'}</div>`;
  }
  function blocksHtml(blocks, prefix, O, from = 0) {
    return (blocks || []).map((b, j) => { const i = j + from; return b.t === 'tbl' ? tblHtml(b, `${prefix}t${i}`, O) : pHtml(b, `${prefix}p${i}`, O); }).join('');
  }
  /* skip: 공통 서식에서 위쪽 제목·결재 줄을 빼고 그린다(그 자리는 headHtml) */
  function tblHtml(t, tp, O, skip = 0) {
    const cols = t.cols.map((w) => `${(+w).toFixed(1)}px`).join(' ');
    const rows = t.rows.slice(skip).map((h) => `minmax(${Math.max(14, +h).toFixed(1)}px,auto)`).join(' ');
    const bw = (n) => (n ? `${n >= 2 ? '2px' : '1px'} solid #111` : '0');
    const cells = t.cells.filter((c) => c.r >= skip).map((c) => {
      const st = [`grid-column:${c.c + 1}/span ${c.cs}`, `grid-row:${c.r - skip + 1}/span ${c.rs}`, `border-left:${bw(c.b.l)}`, `border-right:${bw(c.b.r)}`, `border-top:${bw(c.b.t)}`, `border-bottom:${bw(c.b.b)}`,
        `justify-content:${c.va === 'top' ? 'flex-start' : c.va === 'bottom' ? 'flex-end' : 'center'}`];
      if (c.fill) st.push(`background:${c.fill}`);
      return `<div class="htc" style="${st.join(';')}">${blocksHtml(c.body, `${tp}.r${c.r}c${c.c}.`, O)}</div>`;
    }).join('');
    return `<div class="htbl" style="grid-template-columns:${cols};grid-template-rows:${rows}">${cells}</div>`;
  }

  function sigPad(canvas, onChange) {
    const ctx = canvas.getContext('2d');
    let drawing = false, dirty = false, last = null;
    const size = () => {
      const r = canvas.getBoundingClientRect(), dpr = Math.min(2, root.devicePixelRatio || 1);
      if (!r.width) return;
      const keep = dirty ? canvas.toDataURL('image/png') : '';
      canvas.width = Math.round(r.width * dpr); canvas.height = Math.round(r.height * dpr);
      ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
      ctx.lineWidth = 2.6; ctx.lineCap = 'round'; ctx.lineJoin = 'round'; ctx.strokeStyle = '#111';
      if (keep) { const im = new Image(); im.onload = () => ctx.drawImage(im, 0, 0, r.width, r.height); im.src = keep; }
    };
    const pt = (e) => { const r = canvas.getBoundingClientRect(); return [e.clientX - r.left, e.clientY - r.top]; };
    canvas.style.touchAction = 'none';
    canvas.addEventListener('pointerdown', (e) => { drawing = true; last = pt(e); try { canvas.setPointerCapture(e.pointerId); } catch { /* 일부 브라우저 */ } ctx.beginPath(); ctx.arc(last[0], last[1], 1.2, 0, Math.PI * 2); ctx.fillStyle = '#111'; ctx.fill(); dirty = true; });
    canvas.addEventListener('pointermove', (e) => {
      if (!drawing) return;
      const p = pt(e);
      ctx.beginPath(); ctx.moveTo(last[0], last[1]); ctx.lineTo(p[0], p[1]); ctx.stroke();
      last = p; dirty = true;
    });
    const end = () => { if (!drawing) return; drawing = false; if (onChange) onChange(api.value()); };
    canvas.addEventListener('pointerup', end); canvas.addEventListener('pointercancel', end); canvas.addEventListener('pointerleave', end);
    const api = {
      value: () => (dirty ? canvas.toDataURL('image/png') : ''),
      clear: () => { ctx.clearRect(0, 0, canvas.width, canvas.height); dirty = false; if (onChange) onChange(''); },
      load: (src) => {
        if (!imgOk(src)) return;
        const im = new Image();
        im.onload = () => { const r = canvas.getBoundingClientRect(); ctx.drawImage(im, 0, 0, r.width, r.height); dirty = true; };
        im.src = src;
      },
      size,
    };
    size();
    return api;
  }

  function touched(k) {
    if (k && bad.delete(k)) {
      errs = errs.filter((e) => e.k !== k);
      const el = document.querySelector(`[data-k="${CSS.escape(k)}"]`);
      if (el) el.classList.remove('bad');
      const box = document.querySelector('.errbox');
      if (box && !errs.length) box.remove();

    clearTimeout(saveTimer);
    saveTimer = setTimeout(save, 600);

  function prune(d) {
    if (d.kind !== 'field-report') return;
    const f = d.f;
    const want = new Set(f.unit === 'half' ? [f.halfDate] : SF.weekdays(f.from, f.to));
    f.days = (f.days || []).filter((x) => want.has(x.date));

  async function submit() {
    const d = cur;
    prune(d);
    errs = SF.problems(view(d));
    bad = new Set(errs.map((e) => e.k));
    if (errs.length) {
      redraw(); window.scrollTo(0, 0);
      toast(`채울 곳이 ${errs.length}군데 있어요`);
      return;
    }
    const btn = $('#eSubmit'); btn.disabled = true; btn.textContent = '보내는 중…';
    try {
      const j = await api({ a: 'submit', ...auth(), kind: d.kind, id: d.sid || '', edit: d.edit || '', req: d.req ? d.req.id : '', data: d.f, meta: SF.meta(view(d)) });
      d.sid = j.id; if (j.edit) d.edit = j.edit;
      d.status = 'sent'; d.dirty = false; d.sentAt = j.up;
      save();
      toast('제출했어요. 선생님이 인쇄하기 전까지 고칠 수 있어요.');
      cur = null; go('home');
      refresh(true);
    } catch (e) {
      if (AUTH_ERR.includes(e.code)) { P.me = null; save(); cur = null; go('home'); login(e.message); return; }
      if (e.code === 'locked') { d.status = 'locked'; d.sum = SF.summary(view(d)); delete d.f; delete d.edit; save(); cur = null; go('home'); }
      else if (e.code === 'gone') { d.sid = ''; d.edit = ''; d.status = 'draft'; save(); redraw(); }
      else { btn.disabled = false; btn.textContent = d.status === 'sent' ? '다시 제출' : '제출'; }
      toast(e.message);
    }
  }

    if (id === 'cAgain') { go('home'); return; }
    if (id === 'lgGo') { doLogin(); return; }
    if (id === 'hRefresh') { refresh(false); return; }
    if (id === 'hApply') { cur = newDoc('field-apply', null); errs = []; bad = new Set(); go('edit'); return; }
    if (id === 'hNotMe' && confirm('다른 학생으로 다시 확인할까요? 이 폰에서 쓰던 서류는 지워져요.')) { P.me = null; P.docs = []; save(); go('home'); return; }
    if (id === 'hWipe' && confirm('동의를 철회하고 이 폰의 서류를 모두 지울까요? 이미 낸 서류는 선생님께 지워 달라고 말씀드려 주세요.')) { P = { ...blank(), cfg: P.cfg }; save(); go('home'); return; }
    if (t.dataset.req) {
      const r = reqs.find((x) => x.id === t.dataset.req);
      if (!r) return;
      cur = P.docs.find((d) => d.req && d.req.id === r.id) || newDoc(r.kind, r);
      errs = []; bad = new Set(); go('edit'); return;
    }
    if (t.dataset.open) { cur = P.docs.find((d) => d.lid === t.dataset.open) || null; errs = []; bad = new Set(); go('edit'); return; }
    if (!cur) return;
    if (id === 'eBack') { save(); cur = null; go('home'); return; }
    if (id === 'eSave') { if (save()) toast(cur.status === 'sent' ? '폰에 저장했어요. [다시 제출]해야 선생님께 가요.' : '폰에 저장했어요. 아직 선생님께 안 갔어요.'); return; }
    if (id === 'eSubmit') { submit(); return; }
    if (id === 'ePreview') { preview(); return; }
    if (id === 'eDel' && confirm('이 서류를 지울까요?')) { P.docs = P.docs.filter((d) => d !== cur); save(); cur = null; go('home'); return; }
    if (t.dataset.sigx) { const p = pads.find(([k]) => k === t.dataset.sigx); if (p) p[1].clear(); return; }
    if (t.dataset.seg) {
      const k = t.dataset.seg, v = t.dataset.v;
      if (k === 'lead') cur.f.proxy = v === 'proxy';
      else set(cur.f, k, v);
      if (k === 'kind' || k === 'att') { cur.f.proofs = []; if (k === 'kind' && v !== '출석인정') cur.f.sub = ''; }
      touched(k); redraw();
    }
  });
  document.addEventListener('input', (e) => {
    if (!cur) return;
    const el = e.target;
    if (el.dataset.f) { set(cur.f, el.dataset.f, el.value); touched(el.dataset.f); }
    if (el.dataset.day) {
      const days = cur.f.days || (cur.f.days = []);
      let x = days.find((y) => y.date === el.dataset.day);
      if (!x) days.push(x = { date: el.dataset.day, text: '', photo: '' });
      x.text = el.value; touched(`day.${el.dataset.day}`);
    }
  });
  document.addEventListener('change', async (e) => {
    if (!cur) return;
    const el = e.target;
    if (el.dataset.chk) { set(cur.f, el.dataset.chk, el.checked); touched(el.dataset.chk); }
    if (el.dataset.multi) {
      const k = el.dataset.multi, list = new Set(get(cur.f, k) || []);
      el.checked ? list.add(el.dataset.v) : list.delete(el.dataset.v);
      set(cur.f, k, [...list]); touched(k); redraw();
    }
    if (el.dataset.f && el.type === 'date') redraw();
    if (el.dataset.photo && el.files && el.files[0]) {
      const dt = el.dataset.photo;
      try {
        toast('사진을 줄이는 중…');
        const img = await SF.shrinkPhoto(el.files[0], 1024, 0.7);
        const days = cur.f.days || (cur.f.days = []);
        let x = days.find((y) => y.date === dt);
        if (!x) days.push(x = { date: dt, text: '', photo: '' });
        x.photo = img; touched(`photo.${dt}`); save(); redraw();
        toast('사진을 넣었어요');
      } catch (err) { toast(err.message || '사진을 넣지 못했어요'); }
    }
  });
  document.addEventListener('keydown', (e) => { if (e.key === 'Enter' && (e.target.id === 'lgName' || e.target.id === 'lgNo')) doLogin(); });
  window.addEventListener('pagehide', () => { if (cur) save(); });
  window.addEventListener('resize', () => pads.forEach(([, p]) => p.size()));

  readLink();
  render();
  if (P.cfg && P.consent && P.me) refresh(true);
})();
